\begin{multicols}{2}
\textbf{a / an}\\
\textbf{a} — undosh \emph{tovush} oldidan: a book, a \textbf{u}niversity /ju:/\\
\textbf{an} — unli \emph{tovush} oldidan: an apple, an \textbf{h}our /au/

\smallskip
\textbf{Plurals (ko'plik)}\\
+s: pen $\to$ pens \quad +es (s, x, sh, ch, o): box $\to$ boxes\\
undosh+y $\to$ ies: baby $\to$ babies \quad f/fe $\to$ ves: knife $\to$ knives\\
Noto'g'ri: man$\to$men, child$\to$children, mouse$\to$mice, person$\to$people, sheep$\to$sheep

\columnbreak
\textbf{Personal pronouns}\\
\begin{tabular}{@{}ll@{\qquad}ll@{}}
I & me & we & us\\
you & you & you & you\\
he / she / it & him / her / it & they & them
\end{tabular}\\
(1-ustun — ega, 2-ustun — to'ldiruvchi)

\smallskip
\textbf{to be (Present)}\\
I \textbf{am} \quad he/she/it \textbf{is} \quad we/you/they \textbf{are}\\
Inkor: I'm not, isn't, aren't \quad So'roq: \textbf{Is} he...? \textbf{Are} you...?
\end{multicols}
\end{tcolorbox}
